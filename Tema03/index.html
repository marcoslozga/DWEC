function ejercicio3() {

  console.log('"5" + 2 =', "5" + 2);

  console.log("true + 1 =", true + 1);

  console.log('"20" / "4" =', "20" / "4");

  console.log("0 == false →", 0 == false);
  console.log("0 === false →", 0 === false);
  console.log("null == undefined →", null == undefined);

Ejercicio 4

Plantillas de cadena y error de const

function ejercicio4() {
  console.log("--- Ejercicio 4 · Plantillas de cadena ---");

  const nombre = "Marcos";
  const ciclo = "DAW";
  const curso = "2026/2027";
  const aficion = "Programación";
  let horasEstudio = 6;
  horasEstudio += 2;

  const plantilla = `Me llamo Marcos ${nombre}, estudio daw ${ciclo} superior (${curso}), me gusta la Programación ${aficion} y he estudiado bachillerato ${horasEstudio}h.`;
  alert(plantilla);
  console.log("Template string:", plantilla);

  const concatenado = "Me llamo " + nombre + ", estudio " + ciclo + " (" + curso + "), me gusta la " + aficion + " y he estudiado " + horasEstudio + "h.";
  console.log("Concatenación +:", concatenado);

  console.log("¿Son estrictamente iguales?:", plantilla === concatenado);
}

Error obtenido en consola al intentar reasignar una constante:

Uncaught TypeError: Assignment to constant variable.

Explicación: Ocurre porque las variables declaradas con const no permiten ser reasignadas tras su inicialización.
